
 <%*

// --- 这是用于在控制台进行一次性测试的代码 ---

// 定义配置变量，你可以根据你的实际情况修改
const sourceFolder = "daily"; // 需要监听的日记文件夹
const destinationFolder = "inbox";     // 移动的目标文件夹
const dailyNoteRegex = /^\d{4}-\d{2}-\d{2}\.md$/; // 匹配原始日记文件名的格式

// 注册一个一次性的文件重命名监听器
app.vault.on('rename', (file, oldPath) => {
    console.log("文件重命名事件触发！"); // 在控制台打印日志，证明事件被捕捉
    console.log("新文件对象:", file);
    console.log("旧文件路径:", oldPath);

    const oldFileName = oldPath.split('/').pop(); // 从旧路径中提取完整文件名

    // 开始验证逻辑
    const isInSourceFolder = oldPath.startsWith(sourceFolder);
    const isDailyNoteFormat = dailyNoteRegex.test(oldFileName);

    console.log(`是否在源文件夹(${sourceFolder})中?`, isInSourceFolder);
    console.log(`旧文件名(${oldFileName})是否符合日记格式?`, isDailyNoteFormat);
    
    // 如果文件在指定的源文件夹中，并且旧文件名符合日记格式
    if (isInSourceFolder && isDailyNoteFormat) {
        
        const newPath = `${destinationFolder}/${file.name}`;
        console.log(`准备移动文件到: ${newPath}`);

        // 执行移动操作
        app.vault.rename(file, newPath)
            .then(() => {
                // 弹出Obsidian内部通知，并打印成功日志
                new Notice(`测试成功：笔记已移至 ${destinationFolder}`);
                console.log("文件移动成功！");
            })
            .catch((e) => {
                new Notice(`测试失败: ${e.message}`);
                console.error("文件移动失败:", e);
            });
    } else {
        console.log("条件不满足，不执行移动操作。");
    }
});

// 在控制台打印一条消息，确认监听器已经成功设置
console.log("[测试] 重命名事件监听器已激活。请尝试重命名一个日记文件。");
new Notice("测试脚本已激活，请重命名日记文件。");
 %>
