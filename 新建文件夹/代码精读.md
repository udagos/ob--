/**
 * @brief 初始化主通道传感器节点列表
 *        根据流程编号获取相应传感器节点列表，并计算节点数量
 *
 * @param TransferPara
 * @param FlowNodeList
 *
 * @return
 */
static U32 FlowNodeListInitConn(const TRANSFER_CONFIG *TransferPara, GATE_INDEX GateNo)
{
    BRAN_NODE_LIST *BranNodeList = FlowGetBranNodeList(GateNo);
    FLOW_NODE_LIST *MainFlowNodeList;
    const NODE_ID *TrackNodeList = FlowGetTrackNodelist(TransferPara->mFlowIndex);
    NODE_ID NodeIdInTrack;
    U8 TrackNodeStart = (U8)TRACK_NODE_NUM_MAX;
    U8 NodeCnt = 0;

    if(TransferPara->mFlowIndex >= FLOW_NUM)
    {
        return MakeErrCodeFlow(ERR_FLOW_NODE_LIST, 0);
    }

    if (!IsSrcGate(GateNo))    //  该函数只处理源钞口
    {
        return (0);
    }

    MainFlowNodeList = &gFlowTrackNodeListTable[GateNo];

    memset(MainFlowNodeList, 0x00, sizeof(FLOW_NODE_LIST));
    memset(MainFlowNodeList->mNodeIdPos, 0xff, TRACK_NODE_NUM_MAX);

    for (NodeCnt = 0; NodeCnt < BranNodeList->mTrackNodeNum; NodeCnt++)
    {
        NodeIdInTrack = BranNodeList->mNodeArray[NodeCnt];
        MainFlowNodeList->mNodeArray[NodeCnt] = NodeIdInTrack;
        MainFlowNodeList->mNodeIdPos[NodeIdInTrack] = NodeCnt;
    }

    for (NodeCnt = 0; NodeCnt < TRACK_NODE_NUM_MAX; NodeCnt++)
    {
        NodeIdInTrack = TrackNodeList[NodeCnt];
        if (NodeIdInTrack == BranNodeList->mOverlapNodeId)
        {
            TrackNodeStart = NodeCnt + 1;
            break;
        }
        else if (NodeIdInTrack == NODE_NULL)
        {
            break;
        }
    }

    if (TrackNodeStart >= TRACK_NODE_NUM_MAX)
    {
        return MakeErrCodeFlow(ERR_FLOW_NODE_LIST, 1);
    }

    for(NodeCnt = 0; NodeCnt < TRACK_NODE_NUM_MAX; NodeCnt++)
    {
        NodeIdInTrack = TrackNodeList[NodeCnt + TrackNodeStart];
        if(NodeIdInTrack == NODE_NULL) break;
        MainFlowNodeList->mNodeArray[NodeCnt + BranNodeList->mTrackNodeNum] = NodeIdInTrack;
        MainFlowNodeList->mNodeIdPos[NodeIdInTrack] = NodeCnt + BranNodeList->mTrackNodeNum;
    }
    MainFlowNodeList->mTrackNodeNum = BranNodeList->mTrackNodeNum + NodeCnt;

    if(MainFlowNodeList->mTrackNodeNum >= FLOW_USED_NODE_NUM_MAX)
    {
        DEBUG_LOG_ERR(DEBUG_LOG_FLOW,("FlowNodeNum:%d,PaperMax:%d\n",MainFlowNodeList->mTrackNodeNum,FLOW_USED_NODE_NUM_MAX));
        return MakeErrCodeFlow(ERR_FLOW_NODE_LIST, 2);
    }

    for(NodeCnt = 0; NodeCnt < MainFlowNodeList->mTrackNodeNum; NodeCnt++)
    {
        // PaperEdgeInit(MainFlowNodeList->mNodeArray[NodeCnt]);
        if(NodeCnt < (MainFlowNodeList->mTrackNodeNum - 1))
        {
            MainFlowNodeList->mNodeDisMM[NodeCnt] = (U16)TrackNodeGetDisMM(MainFlowNodeList->mNodeArray[NodeCnt], MainFlowNodeList->mNodeArray[NodeCnt+1]);
            MainFlowNodeList->mNodeDisStep[NodeCnt] = (U16)TrackNodeGetDisStep(MainFlowNodeList->mNodeArray[NodeCnt], MainFlowNodeList->mNodeArray[NodeCnt+1], gSysConfigInfo.mMainMotorResolution);
        }
        else
        {
            MainFlowNodeList->mNodeDisMM[NodeCnt] = 100;
            MainFlowNodeList->mNodeDisStep[NodeCnt] = (U16)(100/gSysConfigInfo.mMainMotorResolution);
        }
    }

    for(NodeCnt = 0; NodeCnt < MainFlowNodeList->mTrackNodeNum; NodeCnt++)
    {
        if (MainFlowNodeList->mNodeArray[NodeCnt] > TRACK_NODE_NUM_MAX)
        {
            TrackNodeUseTwiceFlgSet(MainFlowNodeList->mNodeArray[NodeCnt]);
        }

        // PaperEdgeInit(BranNodeList->mNodeArray[NodeCnt]);
        DEBUG_LOG_TRACE(DEBUG_LOG_FLOW,("MainListConn,Gate:%s,Nd:%s,SN:%d,Dis2Next:%dmm,Step:%d\n",
                    GetGateName(GateNo),
                    TrackNodeGetName(MainFlowNodeList->mNodeArray[NodeCnt]),
                    MainFlowNodeList->mNodeIdPos[MainFlowNodeList->mNodeArray[NodeCnt]],
                    (U16)(MainFlowNodeList->mNodeDisMM[NodeCnt]),
                    MainFlowNodeList->mNodeDisStep[NodeCnt]
                    ));
    }
    
    return 0;
}
