ERRCODE_GATE IsGateMotorErrWhenWaitSem(GATE_HANDLE_MANAGER *ptHandleManager)
{
    if (AppIsMotorRunning(ptHandleManager->mMotorId) == STD_FALSE)
    {
        DEBUG_LOG_ERR(DEBUG_LOG_GATEMODULE, ("Gate:%s,Motor %s is not running %d\n", 
                    GetGateName(ptHandleManager->mGateId), GetMotorName(ptHandleManager->mMotorId)));
        return MakeErrCode(ptHandleManager->mErrCodeDevice, ERR_GROUP_GATE, ERR_GATE_MOTORERR, 1);
    }

    if (OsSemWait(ptHandleManager->mptMotorSem, 10) != OS_EOK)
    {
        ptHandleManager->mMotorSemErrCnt++;

        if (ptHandleManager->mMotorSemErrCnt > 10)  //  如果1s没有接收到同步信号量，需要退出流程的途径，防止电机异常后，流程无法退出
        {
            DEBUG_LOG_ERR(DEBUG_LOG_GATEMODULE, ("Gate:%s,Wait Motor %s Sync Sem Err %d\n", 
                    GetGateName(ptHandleManager->mGateId), GetMotorName(ptHandleManager->mMotorId), ptHandleManager->mMotorSemErrCnt));
            return MakeErrCode(ptHandleManager->mErrCodeDevice, ERR_GROUP_GATE, ERR_GATE_MOTORERR, 0);
        }
        else
        {
            //  电机异常在电机处理线程中进行处理，此处只记录
            DEBUG_LOG_ERR(DEBUG_LOG_GATEMODULE, ("Gate:%s,Wait Motor %s Sync Sem TimeOut %d\n", 
                    GetGateName(ptHandleManager->mGateId), GetMotorName(ptHandleManager->mMotorId), ptHandleManager->mMotorSemErrCnt));
        }
    }

    return ERR_GATE_NOERROR;
}
