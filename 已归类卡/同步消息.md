# RT-Thread 同步消息实现方案：消息队列+信号量确认机制

下面我将提供一个完整的同步消息实现方案，结合消息队列和信号量来实现带确认机制的消息传递。

## 方案一：基本确认模式（请求-确认）

```c
#include <rtthread.h>

/* 定义消息结构 */
struct sync_msg {
    int msg_id;
    int data;
    rt_sem_t ack_sem;  // 用于确认的信号量
};

#define MSG_SIZE sizeof(struct sync_msg)
#define MAX_MSGS 5

static rt_mq_t msg_queue;

/* 发送线程 */
static void sender_thread_entry(void *parameter)
{
    int msg_count = 0;
    
    while (1)
    {
        /* 创建消息和确认信号量 */
        struct sync_msg msg;
        msg.msg_id = msg_count++;
        msg.data = rt_tick_get() % 100;
        msg.ack_sem = rt_sem_create("ack_sem", 0, RT_IPC_FLAG_FIFO);
        
        /* 发送消息 */
        if (rt_mq_send(msg_queue, &msg, MSG_SIZE) == RT_EOK)
        {
            rt_kprintf("Sender: sent msg %d (data=%d), waiting for ack...\n", 
                      msg.msg_id, msg.data);
            
            /* 等待接收方确认 */
            if (rt_sem_take(msg.ack_sem, 1000) == RT_EOK)
            {
                rt_kprintf("Sender: received ack for msg %d\n", msg.msg_id);
            }
            else
            {
                rt_kprintf("Sender: ack timeout for msg %d\n", msg.msg_id);
            }
        }
        else
        {
            rt_kprintf("Sender: failed to send msg %d\n", msg.msg_id);
        }
        
        /* 释放信号量资源 */
        rt_sem_delete(msg.ack_sem);
        
        /* 模拟工作间隔 */
        rt_thread_mdelay(800);
    }
}

/* 接收线程 */
static void receiver_thread_entry(void *parameter)
{
    struct sync_msg msg;
    
    while (1)
    {
        /* 接收消息 */
        if (rt_mq_recv(msg_queue, &msg, MSG_SIZE, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Receiver: got msg %d (data=%d), processing...\n", 
                      msg.msg_id, msg.data);
            
            /* 模拟处理耗时 */
            rt_thread_mdelay(300);
            
            /* 发送确认 */
            rt_sem_release(msg.ack_sem);
            rt_kprintf("Receiver: sent ack for msg %d\n", msg.msg_id);
        }
    }
}

int sync_msg_sample(void)
{
    /* 创建消息队列 */
    msg_queue = rt_mq_create("sync_mq", MSG_SIZE, MAX_MSGS, RT_IPC_FLAG_FIFO);
    if (msg_queue == RT_NULL)
    {
        rt_kprintf("Failed to create message queue!\n");
        return -1;
    }
    
    /* 创建发送线程 */
    rt_thread_t sender = rt_thread_create("sender",
                                       sender_thread_entry,
                                       RT_NULL,
                                       512,
                                       15,
                                       10);
    if (sender != RT_NULL)
    {
        rt_thread_startup(sender);
    }
    
    /* 创建接收线程 */
    rt_thread_t receiver = rt_thread_create("receiver",
                                          receiver_thread_entry,
                                          RT_NULL,
                                          512,
                                          16,
                                          10);
    if (receiver != RT_NULL)
    {
        rt_thread_startup(receiver);
    }
    
    return 0;
}
```

## 方案二：双向通信模式（请求-响应）

```c
#include <rtthread.h>

/* 定义消息类型 */
enum msg_type {
    MSG_REQUEST,
    MSG_RESPONSE
};

/* 定义消息结构 */
struct sync_msg {
    enum msg_type type;
    int request_id;
    int data;
    rt_sem_t sync_sem;  // 用于同步的信号量
};

#define MSG_SIZE sizeof(struct sync_msg)
#define MAX_MSGS 5

static rt_mq_t request_queue;
static rt_mq_t response_queue;

/* 客户端线程 */
static void client_thread_entry(void *parameter)
{
    int req_id = 0;
    
    while (1)
    {
        /* 创建请求消息 */
        struct sync_msg req;
        req.type = MSG_REQUEST;
        req.request_id = req_id++;
        req.data = rt_tick_get() % 100;
        req.sync_sem = rt_sem_create("sync_sem", 0, RT_IPC_FLAG_FIFO);
        
        /* 发送请求 */
        if (rt_mq_send(request_queue, &req, MSG_SIZE) == RT_EOK)
        {
            rt_kprintf("Client: sent request %d (data=%d)\n", 
                      req.request_id, req.data);
            
            /* 等待响应 */
            if (rt_sem_take(req.sync_sem, 1500) == RT_EOK)
            {
                /* 接收响应 */
                struct sync_msg resp;
                if (rt_mq_recv(response_queue, &resp, MSG_SIZE, 0) == RT_EOK)
                {
                    if (resp.request_id == req.request_id)
                    {
                        rt_kprintf("Client: got response for req %d, result=%d\n",
                                  resp.request_id, resp.data);
                    }
                }
            }
            else
            {
                rt_kprintf("Client: timeout waiting for response to req %d\n",
                          req.request_id);
            }
        }
        
        /* 清理资源 */
        rt_sem_delete(req.sync_sem);
        
        /* 模拟请求间隔 */
        rt_thread_mdelay(1000);
    }
}

/* 服务端线程 */
static void server_thread_entry(void *parameter)
{
    struct sync_msg req;
    
    while (1)
    {
        /* 接收请求 */
        if (rt_mq_recv(request_queue, &req, MSG_SIZE, RT_WAITING_FOREVER) == RT_EOK)
        {
            if (req.type == MSG_REQUEST)
            {
                rt_kprintf("Server: processing request %d (data=%d)\n",
                          req.request_id, req.data);
                
                /* 模拟处理耗时 */
                rt_thread_mdelay(200 + (rt_tick_get() % 300));
                
                /* 准备响应 */
                struct sync_msg resp;
                resp.type = MSG_RESPONSE;
                resp.request_id = req.request_id;
                resp.data = req.data * 2;  // 简单处理：数据翻倍
                
                /* 发送响应 */
                if (rt_mq_send(response_queue, &resp, MSG_SIZE) == RT_EOK)
                {
                    /* 通知客户端响应已就绪 */
                    rt_sem_release(req.sync_sem);
                    rt_kprintf("Server: sent response for req %d\n", 
                              req.request_id);
                }
            }
        }
    }
}

int sync_req_resp_sample(void)
{
    /* 创建请求队列 */
    request_queue = rt_mq_create("req_mq", MSG_SIZE, MAX_MSGS, RT_IPC_FLAG_FIFO);
    if (request_queue == RT_NULL)
    {
        rt_kprintf("Failed to create request queue!\n");
        return -1;
    }
    
    /* 创建响应队列 */
    response_queue = rt_mq_create("resp_mq", MSG_SIZE, MAX_MSGS, RT_IPC_FLAG_FIFO);
    if (response_queue == RT_NULL)
    {
        rt_kprintf("Failed to create response queue!\n");
        return -1;
    }
    
    /* 创建客户端线程 */
    rt_thread_t client = rt_thread_create("client",
                                        client_thread_entry,
                                        RT_NULL,
                                        512,
                                        15,
                                        10);
    if (client != RT_NULL)
    {
        rt_thread_startup(client);
    }
    
    /* 创建服务端线程 */
    rt_thread_t server = rt_thread_create("server",
                                        server_thread_entry,
                                        RT_NULL,
                                        512,
                                        16,
                                        10);
    if (server != RT_NULL)
    {
        rt_thread_startup(server);
    }
    
    return 0;
}
```

## 方案三：带超时和重试的可靠同步消息

```c
#include <rtthread.h>

/* 定义消息结构 */
struct reliable_msg {
    int msg_id;
    int data;
    int retry_count;
    rt_sem_t ack_sem;
    rt_tick_t timeout;
};

#define MSG_SIZE sizeof(struct reliable_msg)
#define MAX_MSGS 5
#define MAX_RETRIES 3
#define ACK_TIMEOUT 1000  // 1秒超时

static rt_mq_t reliable_queue;

/* 发送线程 */
static void reliable_sender_entry(void *parameter)
{
    int msg_id = 0;
    
    while (1)
    {
        /* 创建消息 */
        struct reliable_msg msg;
        msg.msg_id = msg_id++;
        msg.data = rt_tick_get() % 100;
        msg.retry_count = 0;
        msg.ack_sem = rt_sem_create("ack_sem", 0, RT_IPC_FLAG_FIFO);
        msg.timeout = ACK_TIMEOUT;
        
        /* 发送循环（带重试） */
        rt_bool_t ack_received = RT_FALSE;
        
        while (msg.retry_count < MAX_RETRIES && !ack_received)
        {
            /* 发送消息 */
            if (rt_mq_send(reliable_queue, &msg, MSG_SIZE) == RT_EOK)
            {
                rt_kprintf("Sender: sent msg %d (try %d), waiting for ack...\n",
                          msg.msg_id, msg.retry_count + 1);
                
                /* 等待确认 */
                if (rt_sem_take(msg.ack_sem, msg.timeout) == RT_EOK)
                {
                    rt_kprintf("Sender: received ack for msg %d\n", msg.msg_id);
                    ack_received = RT_TRUE;
                }
                else
                {
                    rt_kprintf("Sender: ack timeout for msg %d\n", msg.msg_id);
                    msg.retry_count++;
                    msg.timeout += 500; // 每次重试增加等待时间
                }
            }
            else
            {
                rt_kprintf("Sender: queue full, retrying msg %d\n", msg.msg_id);
                rt_thread_mdelay(100);
            }
        }
        
        if (!ack_received)
        {
            rt_kprintf("Sender: giving up on msg %d after %d retries\n",
                      msg.msg_id, MAX_RETRIES);
        }
        
        /* 清理资源 */
        rt_sem_delete(msg.ack_sem);
        
        /* 模拟工作间隔 */
        rt_thread_mdelay(1500);
    }
}

/* 接收线程 */
static void reliable_receiver_entry(void *parameter)
{
    struct reliable_msg msg;
    
    while (1)
    {
        /* 接收消息 */
        if (rt_mq_recv(reliable_queue, &msg, MSG_SIZE, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Receiver: got msg %d (data=%d), processing...\n",
                      msg.msg_id, msg.data);
            
            /* 模拟随机处理失败 (20%概率) */
            if ((rt_tick_get() % 5) == 0)
            {
                rt_kprintf("Receiver: simulated processing failure for msg %d\n",
                          msg.msg_id);
                rt_thread_mdelay(200);
                continue; // 不发送ACK，触发发送方重试
            }
            
            /* 模拟处理耗时 */
            rt_thread_mdelay(100 + (rt_tick_get() % 400));
            
            /* 发送确认 */
            rt_sem_release(msg.ack_sem);
            rt_kprintf("Receiver: sent ack for msg %d\n", msg.msg_id);
        }
    }
}

int reliable_msg_sample(void)
{
    /* 创建消息队列 */
    reliable_queue = rt_mq_create("reliable_mq", MSG_SIZE, MAX_MSGS, RT_IPC_FLAG_FIFO);
    if (reliable_queue == RT_NULL)
    {
        rt_kprintf("Failed to create message queue!\n");
        return -1;
    }
    
    /* 创建发送线程 */
    rt_thread_t sender = rt_thread_create("reliable_sender",
                                        reliable_sender_entry,
                                        RT_NULL,
                                        512,
                                        15,
                                        10);
    if (sender != RT_NULL)
    {
        rt_thread_startup(sender);
    }
    
    /* 创建接收线程 */
    rt_thread_t receiver = rt_thread_create("reliable_receiver",
                                          reliable_receiver_entry,
                                          RT_NULL,
                                          512,
                                          16,
                                          10);
    if (receiver != RT_NULL)
    {
        rt_thread_startup(receiver);
    }
    
    return 0;
}
```

## 方案四：多客户端单服务端模式

```c
#include <rtthread.h>

#define CLIENT_COUNT 3
#define MSG_SIZE sizeof(struct client_msg)
#define MAX_MSGS 10

/* 定义消息结构 */
struct client_msg {
    int client_id;
    int msg_id;
    int data;
    rt_sem_t ack_sem;
};

static rt_mq_t server_queue;
static rt_mq_t client_queues[CLIENT_COUNT];

/* 客户端线程 */
static void client_entry(void *parameter)
{
    int client_id = (int)parameter;
    int msg_id = 0;
    
    while (1)
    {
        /* 创建消息 */
        struct client_msg msg;
        msg.client_id = client_id;
        msg.msg_id = msg_id++;
        msg.data = rt_tick_get() % 100;
        msg.ack_sem = rt_sem_create("ack_sem", 0, RT_IPC_FLAG_FIFO);
        
        /* 发送消息到服务端 */
        if (rt_mq_send(server_queue, &msg, MSG_SIZE) == RT_EOK)
        {
            rt_kprintf("Client %d: sent msg %d, waiting for ack...\n",
                      client_id, msg.msg_id);
            
            /* 等待服务端确认 */
            if (rt_sem_take(msg.ack_sem, 2000) == RT_EOK)
            {
                /* 从专用队列获取响应 */
                struct client_msg resp;
                if (rt_mq_recv(client_queues[client_id], &resp, MSG_SIZE, 0) == RT_EOK)
                {
                    rt_kprintf("Client %d: got response for msg %d, result=%d\n",
                              client_id, resp.msg_id, resp.data);
                }
            }
            else
            {
                rt_kprintf("Client %d: timeout waiting for ack\n", client_id);
            }
        }
        
        /* 清理资源 */
        rt_sem_delete(msg.ack_sem);
        
        /* 模拟工作间隔 */
        rt_thread_mdelay(800 + (client_id * 200));
    }
}

/* 服务端线程 */
static void server_entry(void *parameter)
{
    struct client_msg msg;
    
    while (1)
    {
        /* 接收客户端消息 */
        if (rt_mq_recv(server_queue, &msg, MSG_SIZE, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Server: processing msg %d from client %d (data=%d)\n",
                      msg.msg_id, msg.client_id, msg.data);
            
            /* 模拟处理耗时 */
            rt_thread_mdelay(300 + (rt_tick_get() % 500));
            
            /* 准备响应 */
            struct client_msg resp;
            resp.client_id = msg.client_id;
            resp.msg_id = msg.msg_id;
            resp.data = msg.data * msg.data; // 平方计算
            
            /* 发送响应到客户端专用队列 */
            if (rt_mq_send(client_queues[msg.client_id], &resp, MSG_SIZE) == RT_EOK)
            {
                /* 通知客户端响应已就绪 */
                rt_sem_release(msg.ack_sem);
                rt_kprintf("Server: sent response to client %d for msg %d\n",
                          msg.client_id, msg.msg_id);
            }
        }
    }
}

int multi_client_sample(void)
{
    rt_thread_t clients[CLIENT_COUNT];
    char mq_name[RT_NAME_MAX];
    int i;
    
    /* 创建服务端队列 */
    server_queue = rt_mq_create("server_mq", MSG_SIZE, MAX_MSGS, RT_IPC_FLAG_FIFO);
    if (server_queue == RT_NULL)
    {
        rt_kprintf("Failed to create server queue!\n");
        return -1;
    }
    
    /* 创建客户端专用队列 */
    for (i = 0; i < CLIENT_COUNT; i++)
    {
        rt_snprintf(mq_name, RT_NAME_MAX, "client_mq%d", i);
        client_queues[i] = rt_mq_create(mq_name, MSG_SIZE, 1, RT_IPC_FLAG_FIFO);
        if (client_queues[i] == RT_NULL)
        {
            rt_kprintf("Failed to create client %d queue!\n", i);
            return -1;
        }
    }
    
    /* 创建服务端线程 */
    rt_thread_t server = rt_thread_create("server",
                                        server_entry,
                                        RT_NULL,
                                        1024,
                                        15,
                                        10);
    if (server != RT_NULL)
    {
        rt_thread_startup(server);
    }
    
    /* 创建客户端线程 */
    for (i = 0; i < CLIENT_COUNT; i++)
    {
        char name[RT_NAME_MAX];
        rt_snprintf(name, RT_NAME_MAX, "client%d", i);
        
        clients[i] = rt_thread_create(name,
                                    client_entry,
                                    (void *)i,
                                    512,
                                    16,
                                    10);
        if (clients[i] != RT_NULL)
        {
            rt_thread_startup(clients[i]);
        }
    }
    
    return 0;
}
```

## 同步消息设计模式总结

### 1. 确认机制实现方式对比

| 实现方式 | 优点 | 缺点 | 适用场景 |
|----------|------|------|----------|
| **内嵌信号量** | 实现简单，直接关联消息 | 需要动态创建/销毁信号量 | 简单一对一同步 |
| **专用ACK队列** | 资源管理简单 | 需要额外队列 | 多对多通信 |
| **请求-响应队列** | 双向通信清晰 | 实现较复杂 | 需要返回结果的场景 |
| **超时重试** | 提高可靠性 | 增加延迟 | 不可靠通信环境 |

### 2. 消息队列同步最佳实践

1. **消息设计原则**：
   - 包含唯一消息ID用于匹配请求和响应
   - 添加时间戳用于超时计算和性能监控
   - 明确的消息类型字段（请求/响应/通知等）

2. **资源管理**：
   - 为每个客户端创建专用响应队列
   - 使用对象池管理信号量等资源
   - 设置合理的队列大小防止内存耗尽

3. **错误处理**：
   - 检查所有队列操作的返回值
   - 实现消息重传机制
   - 添加死锁检测和恢复逻辑

4. **性能优化**：
   - 批量处理消息减少上下文切换
   - 根据优先级设计多级队列
   - 为关键消息设置独立高优先级队列

5. **调试支持**：
   - 记录消息流经的关键路径
   - 监控队列深度和等待时间
   - 实现消息追踪机制

## 实际应用建议

1. **简单通知场景**：
   - 使用基本消息队列即可
   - 不需要确认机制

2. **可靠传输场景**：
   - 采用方案三的超时重试模式
   - 添加消息持久化支持

3. **复杂交互场景**：
   - 使用方案四的多客户端模式
   - 为每个客户端维护会话状态

4. **高性能场景**：
   - 使用零拷贝消息传递
   - 考虑无锁队列实现
   - 减少内存分配操作

5. **安全关键场景**：
   - 添加消息完整性校验
   - 实现双重确认机制
   - 添加心跳检测

# RT-Thread 内嵌邮箱实现同步消息通信

下面我将详细介绍如何在 RT-Thread 中使用内嵌邮箱 (mailbox) 实现线程间的同步消息通信，包含多种实际应用场景的完整示例。

## 邮箱基础概念

邮箱是 RT-Thread 中一种高效的线程间通信机制，特点包括：
- 固定大小的消息存储空间
- 支持超时等待
- 消息以指针形式传递（节省内存拷贝）
- 内置同步机制

## 方案一：基本邮箱同步通信

```c
#include <rtthread.h>

/* 定义邮箱 */
static struct rt_mailbox mb;
static char mb_pool[128]; /* 邮箱内存池 */

/* 定义消息结构 */
struct msg {
    int type;
    int value;
};

/* 发送线程 */
static void sender_thread_entry(void *parameter)
{
    int count = 0;
    
    while (1)
    {
        /* 动态创建消息 */
        struct msg *msg = rt_malloc(sizeof(struct msg));
        msg->type = 1;
        msg->value = count++;
        
        /* 发送消息到邮箱 */
        if (rt_mb_send(&mb, (rt_uint32_t)msg) == RT_EOK)
        {
            rt_kprintf("Sender: sent msg %d (type=%d, value=%d)\n", 
                      count, msg->type, msg->value);
        }
        else
        {
            rt_kprintf("Sender: mailbox full! Free msg.\n");
            rt_free(msg);
        }
        
        /* 模拟工作间隔 */
        rt_thread_mdelay(500);
    }
}

/* 接收线程 */
static void receiver_thread_entry(void *parameter)
{
    while (1)
    {
        struct msg *msg;
        
        /* 从邮箱接收消息 */
        if (rt_mb_recv(&mb, (rt_uint32_t *)&msg, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Receiver: got msg (type=%d, value=%d)\n", 
                      msg->type, msg->value);
            
            /* 处理完成后释放消息内存 */
            rt_free(msg);
        }
        
        /* 模拟处理耗时 */
        rt_thread_mdelay(300);
    }
}

int mailbox_sample(void)
{
    /* 初始化邮箱 */
    rt_mb_init(&mb,
              "mymbox",             /* 名称 */
              &mb_pool[0],          /* 内存池 */
              sizeof(mb_pool)/4,    /* 消息数量 */
              RT_IPC_FLAG_FIFO);    /* FIFO模式 */
    
    /* 创建发送线程 */
    rt_thread_t sender = rt_thread_create("sender",
                                        sender_thread_entry,
                                        RT_NULL,
                                        512,
                                        15,
                                        10);
    if (sender != RT_NULL)
    {
        rt_thread_startup(sender);
    }
    
    /* 创建接收线程 */
    rt_thread_t receiver = rt_thread_create("receiver",
                                          receiver_thread_entry,
                                          RT_NULL,
                                          512,
                                          16,
                                          10);
```

## 方案二：带确认机制的邮箱同步

```c
#include <rtthread.h>

/* 定义两个邮箱 */
static struct rt_mbox request_mb, response_mb;
static char req_pool[128], resp_pool[64]; /* 邮箱内存池 */

/* 定义消息结构 */
struct req_msg {
    int req_id;
    int data;
    rt_uint32_t sender_tid; /* 发送方线程ID */
};

struct resp_msg {
    int req_id;
    int result;
};

/* 客户端线程 */
static void client_thread_entry(void *parameter)
{
    int req_count = 0;
    
    while (1)
    {
        /* 准备请求 */
        struct req_msg *req = rt_malloc(sizeof(struct req_msg));
        req->req_id = req_count++;
        req->data = rt_tick_get() % 100;
        req->sender_tid = (rt_uint32_t)rt_thread_self();
        
        /* 发送请求 */
        if (rt_mb_send(&request_mb, (rt_uint32_t)req) == RT_EOK)
        {
            rt_kprintf("Client: sent req %d (data=%d)\n", 
                      req->req_id, req->data);
            
            /* 等待响应 */
            struct resp_msg *resp;
            if (rt_mb_recv(&response_mb, (rt_uint32_t *)&resp, 1000) == RT_EOK)
            {
                if (resp->req_id == req->req_id)
                {
                    rt_kprintf("Client: got resp for req %d, result=%d\n",
                              resp->req_id, resp->result);
                }
                rt_free(resp);
            }
            else
            {
                rt_kprintf("Client: timeout waiting for resp to req %d\n",
                          req->req_id);
            }
        }
        else
        {
            rt_kprintf("Client: failed to send req %d\n", req->req_id);
            rt_free(req);
        }
        
        /* 模拟请求间隔 */
        rt_thread_mdelay(800);
    }
}

/* 服务端线程 */
static void server_thread_entry(void *parameter)
{
    while (1)
    {
        /* 接收请求 */
        struct req_msg *req;
        if (rt_mb_recv(&request_mb, (rt_uint32_t *)&req, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Server: processing req %d from thread 0x%x (data=%d)\n",
                      req->req_id, req->sender_tid, req->data);
            
            /* 模拟处理耗时 */
            rt_thread_mdelay(200 + (rt_tick_get() % 300));
            
            /* 准备响应 */
            struct resp_msg *resp = rt_malloc(sizeof(struct resp_msg));
            resp->req_id = req->req_id;
            resp->result = req->data * 2; /* 简单处理 */
            
            /* 发送响应 */
            if (rt_mb_send(&response_mb, (rt_uint32_t)resp) != RT_EOK)
            {
                rt_kprintf("Server: failed to send resp for req %d\n", req->req_id);
                rt_free(resp);
            }
            
            /* 释放请求内存 */
            rt_free(req);
        }
    }
}

int mailbox_req_resp_sample(void)
{
    /* 初始化请求邮箱 */
    rt_mb_init(&request_mb,
              "req_mb",
              &req_pool[0],
              sizeof(req_pool)/4,
              RT_IPC_FLAG_FIFO);
    
    /* 初始化响应邮箱 */
    rt_mb_init(&response_mb,
              "resp_mb",
              &resp_pool[0],
              sizeof(resp_pool)/4,
              RT_IPC_FLAG_FIFO);
    
    /* 创建客户端线程 */

                                        RT_NULL,
                                        512,
                                        15,
                                        10);
    if (client != RT_NULL)
    {
        rt_thread_startup(client);
    }
    
    /* 创建服务端线程 */
    rt_thread_t server = rt_thread_create("server",
                                        server_thread_entry,
                                        RT_NULL,
                                        512,
                                        16,
                                        10);
    if (server != RT_NULL)
    {
        rt_thread_startup(server);

    return 0;
}
```

## 方案三：邮箱+信号量实现强同步

```c
#include <rtthread.h>

/* 定义邮箱和信号量 */
static struct rt_mbox sync_mb;
static char mb_pool[64];
static rt_sem_t ack_sem;

/* 定义消息结构 */
struct sync_msg {
    int msg_id;
    int data;
    rt_sem_t reply_sem; /* 用于回复的信号量 */
};

/* 发送线程 */
static void sync_sender_entry(void *parameter)
{
    int msg_id = 0;
    
    while (1)
    {
        /* 创建消息和信号量 */
        struct sync_msg *msg = rt_malloc(sizeof(struct sync_msg));
        msg->msg_id = msg_id++;
        msg->data = rt_tick_get() % 100;
        msg->reply_sem = rt_sem_create("reply_sem", 0, RT_IPC_FLAG_FIFO);
        
        /* 获取发送许可 */
        rt_sem_take(ack_sem, RT_WAITING_FOREVER);
        
        /* 发送消息 */
        if (rt_mb_send(&sync_mb, (rt_uint32_t)msg) == RT_EOK)
        {
            rt_kprintf("Sender: sent sync msg %d (data=%d)\n", 
                      msg->msg_id, msg->data);
            
            /* 等待回复 */
            if (rt_sem_take(msg->reply_sem, 1500) == RT_EOK)
            {
                rt_kprintf("Sender: got reply for msg %d\n", msg->msg_id);
            }
            else
            {
                rt_kprintf("Sender: timeout waiting for reply to msg %d\n",
                          msg->msg_id);
            }
        }
        
        /* 释放资源 */
        rt_sem_delete(msg->reply_sem);
        rt_free(msg);
        
        /* 模拟工作间隔 */
        rt_thread_mdelay(1000);
    }
}

/* 接收线程 */
static void sync_receiver_entry(void *parameter)
{
    while (1)
    {
        /* 接收消息 */
        struct sync_msg *msg;
        if (rt_mb_recv(&sync_mb, (rt_uint32_t *)&msg, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Receiver: got sync msg %d (data=%d)\n",
                      msg->msg_id, msg->data);
            
            /* 模拟处理耗时 */
            rt_thread_mdelay(300 + (rt_tick_get() % 400));
            
            /* 发送回复 */
            rt_sem_release(msg->reply_sem);
            
            /* 允许发送下一个消息 */
            rt_sem_release(ack_sem);
            
            rt_kprintf("Receiver: sent reply for msg %d\n", msg->msg_id);
        }
    }
}

int mailbox_sync_sample(void)
{
    /* 初始化邮箱 */
    rt_mb_init(&sync_mb,
              "sync_mb",
              &mb_pool[0],
              sizeof(mb_pool)/4,
              RT_IPC_FLAG_FIFO);
    
    /* 创建信号量 */
    ack_sem = rt_sem_create("ack_sem", 1, RT_IPC_FLAG_FIFO);
    
    /* 创建发送线程 */
    rt_thread_t sender = rt_thread_create("sync_sender",
                                        sync_sender_entry,
                                        RT_NULL,
                                        512,
                                        15,
                                        10);
    if (sender != RT_NULL)
    {
        rt_thread_startup(sender);
    }
    
    /* 创建接收线程 */
    rt_thread_t receiver = rt_thread_create("sync_receiver",
                                          sync_receiver_entry,
                                          RT_NULL,
                                          512,
                                          16,
                                          10);
    if (receiver != RT_NULL)
    {
        rt_thread_startup(receiver);
    }
    
    return 0;
}
```

## 方案四：多对多邮箱通信

```c
#include <rtthread.h>

#define PRODUCER_COUNT 2
#define CONSUMER_COUNT 3
#define MB_SIZE 5
#define MSG_POOL_SIZE (MB_SIZE * 4)

/* 全局邮箱 */
static struct rt_mbox multi_mb;
static char mb_pool[MB_SIZE * 4]; /* 每个邮箱项占4字节 */

/* 消息结构 */
struct multi_msg {
    int producer_id;
    int msg_id;
    int data;
};

/* 生产者线程 */
static void producer_entry(void *parameter)
{
    int id = (int)parameter;
    int msg_id = 0;
    
    while (1)
    {
        /* 创建消息 */
        struct multi_msg *msg = rt_malloc(sizeof(struct multi_msg));
        msg->producer_id = id;
        msg->msg_id = msg_id++;
        msg->data = rt_tick_get() % 100;
        
        /* 发送消息 */
        if (rt_mb_send(&multi_mb, (rt_uint32_t)msg) == RT_EOK)
        {
            rt_kprintf("Producer %d: sent msg %d (data=%d)\n",
                      id, msg->msg_id, msg->data);
        }
        else
        {
            rt_kprintf("Producer %d: mailbox full! Free msg.\n", id);
            rt_free(msg);
        }
        
        /* 模拟生产间隔 */
        rt_thread_mdelay(400 + (id * 100));
    }
}

/* 消费者线程 */
static void consumer_entry(void *parameter)
{
    int id = (int)parameter;
    
    while (1)
    {
        /* 接收消息 */
        struct multi_msg *msg;
        if (rt_mb_recv(&multi_mb, (rt_uint32_t *)&msg, RT_WAITING_FOREVER) == RT_EOK)
        {
            rt_kprintf("Consumer %d: got msg from producer %d (msg_id=%d, data=%d)\n",
                      id, msg->producer_id, msg->msg_id, msg->data);
            
            /* 模拟处理耗时 */
            rt_thread_mdelay(200 + (id * 50));
            
            /* 释放消息内存 */
            rt_free(msg);
        }
    }
}

int multi_mailbox_sample(void)
{
    rt_thread_t producers[PRODUCER_COUNT];
    rt_thread_t consumers[CONSUMER_COUNT];
    int i;
    
    /* 初始化邮箱 */
    rt_mb_init(&multi_mb,
              "multi_mb",
              &mb_pool[0],
              MB_SIZE,
              RT_IPC_FLAG_FIFO);
    
    /* 创建生产者线程 */
    for (i = 0; i < PRODUCER_COUNT; i++)
    {
        char name[RT_NAME_MAX];
        rt_snprintf(name, RT_NAME_MAX, "producer%d", i);
        
        producers[i] = rt_thread_create(name,
                                      producer_entry,
                                      (void *)i,
                                      512,
                                      15,
                                      10);
        if (producers[i] != RT_NULL)
        {
            rt_thread_startup(producers[i]);
        }
    }
    
    /* 创建消费者线程 */
    for (i = 0; i < CONSUMER_COUNT; i++)
    {
        char name[RT_NAME_MAX];
        rt_snprintf(name, RT_NAME_MAX, "consumer%d", i);
        
        consumers[i] = rt_thread_create(name,
                                      consumer_entry,
                                      (void *)i,
                                      512,
                                      16,
                                      10);
        if (consumers[i] != RT_NULL)
        {
            rt_thread_startup(consumers[i]);
        }
    }
    
    return 0;
}
```

## 邮箱同步模式总结

### 1. 邮箱特性分析

| 特性 | 描述 | 优势 |
|------|------|------|
| **指针传递** | 传递消息指针而非数据拷贝 | 高效，适合大消息 |
| **固定容量** | 预先分配固定数量消息槽 | 内存可控 |
| **超时机制** | 支持发送和接收超时 | 避免永久阻塞 |
| **同步特性** | 内置等待队列 | 自动挂起/唤醒线程 |

### 2. 邮箱同步模式对比

| 模式 | 实现方式 | 适用场景 |
|------|----------|----------|
| **基本异步** | 单纯邮箱发送/接收 | 简单通知，不关心结果 |
| **请求-响应** | 双邮箱 (请求+响应) | 需要返回结果的调用 |
| **强同步** | 邮箱+信号量 | 严格顺序执行 |
| **多对多** | 共享邮箱+消息ID | 多生产者多消费者 |

### 3. 邮箱使用最佳实践

1. **消息生命周期管理**：
   - 发送方分配内存，接收方释放
   - 或者使用全局消息池

2. **错误处理**：
   ```c
   /* 发送失败处理 */
   if (rt_mb_send(mb, (rt_uint32_t)msg) != RT_EOK) {
       rt_free(msg); // 记得释放内存
   }
   
   /* 接收超时处理 */
   if (rt_mb_recv(mb, &msg, 100) == -RT_ETIMEOUT) {
       // 超时处理
   }
   ```

3. **性能优化**：
   - 避免在消息中嵌入大块数据
   - 合理设置邮箱大小（太大浪费内存，太小容易满）
   - 考虑使用静态消息池减少动态分配

4. **调试技巧**：
   - 在消息中添加序列号和时间戳
   - 监控邮箱使用率：
     ```c
     rt_size_t used = rt_mb_get_count(&mb);
     rt_size_t total = rt_mb_get_max_used(&mb);
     ```

### 1. 静态消息池优化

```c
#define MSG_POOL_SIZE 10

/* 静态消息池 */
struct msg msg_pool[MSG_POOL_SIZE];
rt_sem_t msg_sem = rt_sem_create("msg_sem", MSG_POOL_SIZE, RT_IPC_FLAG_FIFO);

/* 获取消息 */
struct msg *alloc_msg(void)
{
    rt_sem_take(msg_sem, RT_WAITING_FOREVER);
    for (int i = 0; i < MSG_POOL_SIZE; i++) {
        if (msg_pool[i].in_use == RT_FALSE) {
            msg_pool[i].in_use = RT_TRUE;
            return &msg_pool[i];
        }
    }
    return NULL; // 不会执行到这里
}

/* 释放消息 */
void free_msg(struct msg *msg)
{
    msg->in_use = RT_FALSE;
    rt_sem_release (msg_sem);
}
```

### 2. 优先级消息处理

```c
enum msg_priority {
    MSG_PRIO_HIGH,
    MSG_PRIO_NORMAL
};

/* 发送优先级消息 */
void send_prio_msg (struct rt_mailbox *mb, enum msg_priority prio, void *msg)
{
    if (prio == MSG_PRIO_HIGH) {
        /* 紧急消息尝试立即发送 */
        if (rt_mb_send (mb, (rt_uint32_t) msg) != RT_EOK) {
            /* 如果邮箱满，移除一个低优先级消息 */
            void *old_msg;
            if (rt_mb_recv (mb, (rt_uint32_t *)&old_msg, 0) == RT_EOK) {
                rt_free (old_msg); // 释放被挤出的消息
                rt_mb_send (mb, (rt_uint32_t) msg); // 重新发送
            }
        }
    } else {
        /* 普通消息正常发送 */
        rt_mb_send (mb, (rt_uint32_t) msg);
    }
}
```
