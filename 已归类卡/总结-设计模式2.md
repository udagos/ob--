### 1.工厂模式
**问题**：如何创建对象而不需要指定具体的类？

**示例**：假设你有一个披萨店，可以制作不同种类的披萨（如芝士披萨、海鲜披萨等）。你希望根据客户的选择创建相应的披萨对象。
```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    void (*draw)(void);
} Shape;

void circle_draw() {
    printf("Drawing Circle\n");
}

void rectangle_draw() {
    printf("Drawing Rectangle\n");
}

Shape* create_shape(const char* type) {
    Shape* shape = (Shape*)malloc(sizeof(Shape));
    if (strcmp(type, "circle") == 0) {
        shape->draw = circle_draw;
    } else if (strcmp(type, "rectangle") == 0) {
        shape->draw = rectangle_draw;
    } else {
        free(shape);
        return NULL;
    }
    return shape;
}

int main() {
    Shape* circle = create_shape("circle");
    if (circle != NULL) {
        circle->draw();  // 输出: Drawing Circle
        free(circle);
    }

    Shape* rectangle = create_shape("rectangle");
    if (rectangle != NULL) {
        rectangle->draw();  // 输出: Drawing Rectangle
        free(rectangle);
    }

    return 0;
}
```

**解决方案**：工厂模式**定义一个接口来创建对象，但由子类决定实例化哪个**类。
###  2.观察者模式
**问题**：如何实现对象之间的一对多依赖关系，当一个对象状态改变时，所有依赖它的对象都会收到通知并自动更新？

**示例**：假设你有一个天气预报系统，当天气数据更新时，所有订阅了天气预报的用户都会收到通知。

**解决方案**：观察者模式定义了一种一对多的依赖关系，当一个对象状态改变时，所有依赖它的对象都会收到通知并自动更新。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct Observer Observer;

typedef struct {
    void (*update)(Observer*);
} Subject;

struct Observer {
    Subject* subject;
    void (*notify)(Observer*);
};

void observer_notify(Observer* observer) {
    observer->subject->update(observer);
}

void subject_update(Observer* observer) {
    printf("Observer notified\n");
}

int main() {
    Subject subject = { .update = subject_update };

    Observer observer1 = { .subject = &subject, .notify = observer_notify };
    Observer observer2 = { .subject = &subject, .notify = observer_notify };

    observer1.notify(&observer1);  // 输出: Observer notified
    observer2.notify(&observer2);  // 输出: Observer notified

    return 0;
}
```

```c
#include <stdio.h>
#include <stdlib.h>

// 观察者接口
typedef struct Observer {
    void (*update)(struct Observer*, const char*);
} Observer;

// 具体观察者
typedef struct ConcreteObserver {
    Observer base;
    char* name;
} ConcreteObserver;

void ConcreteObserverUpdate(Observer* observer, const char* message) {
    ConcreteObserver* concreteObserver = (ConcreteObserver*)observer;
    printf("%s received message: %s\n", concreteObserver->name, message);
}
```


```
// 主题接口
typedef struct Subject {
    Observer** observers;
    size_t observersCount;
    void (*attach)(struct Subject*, Observer*);
    void (*detach)(struct Subject*, Observer*);
    void (*notify)(struct Subject*, const char*);
} Subject;

// 具体主题
typedef struct ConcreteSubject {
    Subject base;
    char* state;
} ConcreteSubject;

void SubjectAttach(Subject* subject, Observer* observer) {
    subject->observers = realloc(subject->observers, sizeof(Observer*) * (subject->observersCount + 1));
    subject->observers[subject->observersCount++] = observer;
}

void SubjectDetach(Subject* subject, Observer* observer) {
    for (size_t i = 0; i < subject->observersCount; ++i) {
        if (subject->observers[i] == observer) {
            for (size_t j = i; j < subject->observersCount - 1; ++j) {
                subject->observers[j] = subject->observers[j + 1];
            }
            subject->observersCount--;
            subject->observers = realloc(subject->observers, sizeof(Observer*) * subject->observersCount);
            break;
        }
    }
}

void SubjectNotify(Subject* subject, const char* message) {
    for (size_t i = 0; i < subject->observersCount; ++i) {
        subject->attach(subject, subject->observers[i]);
        subject->observers[i]->update(subject->observers[i], message);
    }
}
```

```
int main() {
    // 创建具体主题
    ConcreteSubject subject = { {SubjectAttach, SubjectDetach, SubjectNotify}, "initial state" };
    subject.base.observers = NULL;
    subject.base.observersCount = 0;

    // 创建具体观察者
    ConcreteObserver observer1 = { {ConcreteObserverUpdate}, "Observer 1" };
    ConcreteObserver observer2 = { {ConcreteObserverUpdate}, "Observer 2" };

    // 将观察者附加到主题
    subject.base.attach(&subject.base, &observer1.base);
    subject.base.attach(&subject.base, &observer2.base);

    // 状态改变，通知观察者
    subject.state = "new state";
    subject.base.notify(&subject.base, subject.state);

    // 移除观察者
    subject.base.detach(&subject.base, &observer1.base);

    // 状态再次改变，通知剩余的观察者
    subject.state = "final state";
    subject.base.notify(&subject.base, subject.state);

    // 清理内存
    free(subject.base.observers);

    return 0;
}
```

Observer：这是一个观察者接口，包含一个update方法，用于更新观察者的状态。
ConcreteObserver：这是一个具体观察者类，实现了Observer接口的update方法。
Subject：这是一个主题接口，包含attach、detach和notify方法，用于管理观察者。
ConcreteSubject：这是一个具体主题类，实现了Subject接口，并维护了一个观察者列表。
main：客户端代码创建一个具体主题和两个具体观察者，将观察者附加到主题，然后改变主题的状态并通知观察者。最后，移除一个观察者，并再次通知剩余的观察者。
这个示例展示了如何在C语言中实现观察者模式，允许多个观察者监听一个主题的状态变化，并在状态变化时得到通知。

### 3. 策略模式（Strategy Pattern）

**问题**：如何定义一系列算法或策略，并将它们封装在独立的类中，使它们可以互换使用？

**示例**：假设你有一个支付系统，支持多种支付方式（如信用卡支付、支付宝支付、微信支付等）。你希望在运行时动态选择支付方式。

**解决方案**：策略模式定义了一系列算法或策略，并将它们封装在独立的类中，使它们可以互换使用。


以下是一个简单的C语言示例，演示了如何实现策略模式。在这个例子中，我们将创建一个简单的排序程序，其中包含不同的排序策略（如冒泡排序和快速排序）。

 1. 排序策略接口

```c
#include <stdio.h>
#include <stdlib.h>

// 排序策略接口
typedef struct SortStrategy {
    void (*sort)(struct SortStrategy*, int[], int);
} SortStrategy;

// 具体排序策略：冒泡排序
typedef struct BubbleSort {
    SortStrategy base;
} BubbleSort;

void bubbleSort (SortStrategy* strategy, int array[], int size) {
    int i, j;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (array[j] > array[j + 1]) {
                int temp = array[j];
                array[j] = array[j + 1];
                array[j + 1] = temp;
            }
        }
    }
    printf ("Bubble sort: ");
    for (i = 0; i < size; i++) {
        printf ("%d ", array[i]);
    }
    printf ("\n");
}

// 具体排序策略：快速排序
typedef struct QuickSort {
    SortStrategy base;
} QuickSort;

void quickSort (SortStrategy* strategy, int array[], int size) {
    // 简单的快速排序实现
    if (size < 2) return;
    int pivot = array[0];
    int left[100], right[100];
    int l = 0, r = 0;
    for (int i = 1; i < size; i++) {
        if (array[i] < pivot)
            left[l++] = array[i];
        else
            right[r++] = array[i];
    }
    quickSort ((QuickSort*)&left, left, l);
    quickSort ((QuickSort*)&right, right, r);
    
    int writeIndex = 0;
    for (int i = 0; i < l; i++)
        array[writeIndex++] = left[i];
    array[writeIndex++] = pivot;
    for (int i = 0; i < r; i++)
        array[writeIndex++] = right[i];
    
    printf ("Quick sort: ");
    for (i = 0; i < size; i++) {
        printf ("%d ", array[i]);
    }
    printf ("\n");
}

// 创建排序策略实例
SortStrategy* CreateBubbleSort () {
    BubbleSort* bubbleSort = (BubbleSort*) malloc (sizeof (BubbleSort));
    bubbleSort->base. sort = bubbleSort;
    return (SortStrategy*) bubbleSort;
}

SortStrategy* CreateQuickSort () {
    QuickSort* quickSort = (QuickSort*) malloc (sizeof (QuickSort));
    quickSort->base. sort = quickSort;
    return (SortStrategy*) quickSort;
}
```

 2. 客户端代码

```c
int main () {
    int array[] = {3, 6, 8, 10, 1, 2, 1};
    int size = sizeof (array) / sizeof (array[0]);
    
    // 使用冒泡排序策略
    SortStrategy* bubbleSortStrategy = CreateBubbleSort ();
    bubbleSortStrategy->sort (bubbleSortStrategy, array, size);
    free (bubbleSortStrategy);
    
    // 使用快速排序策略
    SortStrategy* quickSortStrategy = CreateQuickSort ();
    quickSortStrategy->sort (quickSortStrategy, array, size);
    free (quickSortStrategy);

    return 0;
}
```

 解释：
- **SortStrategy**：这是一个排序策略接口，包含一个`sort`方法。
- **BubbleSort** 和 **QuickSort**：这些是具体排序策略类，实现了`SortStrategy`接口的`sort`方法。
- **CreateBubbleSort** 和 **CreateQuickSort**：这些函数用于创建具体的排序策略实例。
- **main**：客户端代码创建不同的排序策略实例，并使用它们对同一个数组进行排序。

这个示例展示了如何使用策略模式来封装不同的排序算法，使得它们可以互相替换。策略模式使得算法的变化独立于使用算法的客户，提高了系统的灵活性和可扩展性。

### 4. 装饰器模式（Decorator Pattern）

**问题**：如何在运行时动态地为对象添加功能，而不改变其结构？

**示例**：假设你有一个咖啡店，可以为咖啡添加不同的配料（如牛奶、糖、巧克力等）。你希望在运行时动态地为咖啡添加配料。

**解决方案**：装饰器模式允许你通过将对象包装在装饰器类的实例中来动态地扩展对象的功能。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    float (*cost)(void);
} Beverage;

float coffee_cost() {
    return 5.0;
}

typedef struct {
    Beverage* beverage;
    float (*cost)(void);
} MilkDecorator;

float milk_cost() {
    return 2.0;
}

float milk_decorator_cost(MilkDecorator* decorator) {
    return decorator->beverage->cost() + decorator->cost();
}

int main() {
    Beverage coffee = { .cost = coffee_cost };
    printf("Coffee cost: %.2f\n", coffee.cost());  // 输出: Coffee cost: 5.00

    MilkDecorator milk_decorator = { .beverage = &coffee, .cost = milk_cost };
    printf("Coffee with milk cost: %.2f\n", milk_decorator_cost(&milk_decorator));  // 输出: Coffee with milk cost: 7.00

    return 0;
}
```

Beverage（饮料接口）：**定义了饮料的接口**，包含计算价格的方法cost ()。
Coffee（具体饮料）：**实现了饮料接口**，提供了咖啡的价格计算方法。
MilkDecorator（牛奶装饰器）：**持有一个饮料对象的引用，并实现了饮料接口**。
MilkDecorator（具体牛奶装饰器）：**扩展了装饰器类**，添加了牛奶的价格计算方
### 5. 适配器模式（Adapter Pattern）

**问题**：如何使不兼容的接口能够一起工作？

**示例**：假设你有一个旧的音频播放器，只能播放MP3格式的音频文件。现在你有一个新的音频文件格式（如WAV），你希望旧的播放器能够播放这种新格式的音频文件。

**解决方案**：适配器模式将一个类的接口转换成客户端期望的另一个接口，使原本不兼容的类能够一起工作。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    void (*play_mp3)(const char*);
} MP3Player;

void mp3_player_play_mp3(const char* filename) {
    printf("Playing MP3 file: %s\n", filename);
}

typedef struct {
    void (*play_wav)(const char*);
} WAVPlayer;

void wav_player_play_wav(const char* filename) {
    printf("Playing WAV file: %s\n", filename);
}

typedef struct {
    WAVPlayer* wav_player;
    void (*play_mp3)(const char*);
} WAVToMP3Adapter;

void wav_to_mp3_adapter_play_mp3(const char* filename) {
    printf("Converting WAV file to MP3: %s\n", filename);
    WAVToMP3Adapter* adapter = (WAVToMP3Adapter*)filename;
    adapter->wav_player->play_wav(filename);
}

int main() {
    MP3Player mp3_player = { .play_mp3 = mp3_player_play_mp3 };
    mp3_player.play_mp3("song.mp3");  // 输出: Playing MP3 file: song.mp3

    WAVPlayer wav_player = { .play_wav = wav_player_play_wav };
    WAVToMP3Adapter wav_adapter = { .wav_player = &wav_player, .play_mp3 = wav_to_mp3_adapter_play_mp3 };
    wav_adapter.play_mp3((const char*)&wav_adapter);  // 输出: Converting WAV file to MP3: song.wav
                                                     //       Playing WAV file: song.wav

    return 0;
}
```

### 6. 职责链模式（Chain of Responsibility Pattern）

**问题**：如何将请求的发送者和接收者解耦，并允许多个对象处理请求？

**示例**：假设你有一个支持多种支付方式的系统（如信用卡、支付宝、微信支付等）。你希望将支付请求传递给多个处理器，直到有一个处理器处理成功。

**解决方案**：职责链模式将请求的发送者和接收者解耦，并允许多个对象处理请求，直到有一个对象处理成功。


以下是一个简单的C语言示例，演示了如何实现责任链模式。在这个例子中，我们将创建一个简单的费用报销系统，其中包含多个处理不同费用范围的处理器。

 1. 处理器接口
```c
#include <stdio.h>
#include <stdlib.h>

// 处理器接口
typedef struct Handler {
    void (*handleRequest)(struct Handler*, double);
    struct Handler* nextHandler;
} Handler;

// 构造处理器
void HandlerConstruct (Handler* handler, Handler* next) {
    handler->nextHandler = next;
}

// 处理器请求处理方法
void HandleRequest (Handler* handler, double amount) {
    if (handler == NULL) {
        return;
    }
    handler->handleRequest (handler, amount);
}
```

 2. 具体处理器

```c
// 具体处理器：小额费用处理器
typedef struct SmallAmountHandler {
    Handler base;
} SmallAmountHandler;

void SmallAmountHandlerRequest (Handler* handler, double amount) {
    if (amount <= 100) {
        printf ("Small amount handler approves %0.2f\n", amount);
    } else {
        HandleRequest (handler->nextHandler, amount);
    }
}

// 具体处理器：中等费用处理器
typedef struct MediumAmountHandler {
    Handler base;
} MediumAmountHandler;

void MediumAmountHandlerRequest (Handler* handler, double amount) {
    if (amount > 100 && amount <= 500) {
        printf ("Medium amount handler approves %0.2f\n", amount);
    } else {
        HandleRequest (handler->nextHandler, amount);
    }
}

// 具体处理器：大额费用处理器
typedef struct LargeAmountHandler {
    Handler base;
} LargeAmountHandler;

void LargeAmountHandlerRequest (Handler* handler, double amount) {
    if (amount > 500) {
        printf ("Large amount handler approves %0.2f\n", amount);
    } else {
        printf ("Request of %0.2f is rejected\n", amount);
    }
}
```

 3. 创建处理器链

```c
int main () {
    // 创建具体处理器实例
    LargeAmountHandler largeHandler;
    MediumAmountHandler mediumHandler;
    SmallAmountHandler smallHandler;

    // 构建责任链
    HandlerConstruct (&smallHandler. base, &mediumHandler. base);
    HandlerConstruct (&mediumHandler. base, &largeHandler. base);

    // 设置处理方法
    smallHandler. base. handleRequest = SmallAmountHandlerRequest;
    mediumHandler. base. handleRequest = MediumAmountHandlerRequest;
    largeHandler. base. handleRequest = LargeAmountHandlerRequest;

    // 发送请求
    HandleRequest (&smallHandler. base, 50);   // 小额
    HandleRequest (&smallHandler. base, 200);  // 中等额
    HandleRequest (&smallHandler. base, 600);  // 大额

    return 0;
}
```

 解释：

- **Handler**：这是一个处理器接口，包含一个`handleRequest`方法和一个指向下一个处理器的指针。
- **SmallAmountHandler**、**MediumAmountHandler**、**LargeAmountHandler**：这些是具体的处理器类，实现了`Handler`接口的`handleRequest`方法，并根据金额范围处理请求或将请求转发给下一个处理器。
- **main**：客户端代码创建具体的处理器实例并构建责任链。然后，它发送不同的报销请求，每个请求都会在责任链中依次处理。

这个示例展示了如何使用责任链模式来处理请求，其中每个处理器都有机会处理请求，如果当前处理器不处理该请求，则将请求转发给链中的下一个处理器。这种模式使得请求的发送者和接收者解耦，增强了系统的灵活性。


### 7. 命令模式（Command Pattern）

**问题**：如何将请求封装为对象，从而使你可以用不同的请求、队列或日志来参数化其他对象？

**示例**：创建一个简单的电灯开关系统，使用命令模式来封装“开灯”和“关灯”的请求。



 1. 命令接口
```c
#include <stdio.h>
#include <stdlib.h>

// 命令接口
typedef struct Command {
    void (*execute)(struct Command*);
} Command;

// 接收者
typedef struct Light {
    void (*turnOn)(struct Light*);
    void (*turnOff)(struct Light*);
} Light;

void LightTurnOn (Light* light) {
    printf ("Light is on\n");
}

void LightTurnOff (Light* light) {
    printf ("Light is off\n");
}

// 创建接收者对象
Light* CreateLight () {
    Light* light = (Light*) malloc (sizeof (Light));
    light->turnOn = LightTurnOn;
    light->turnOff = LightTurnOff;
    return light;
}

// 销毁接收者对象
```

```c

    turnOn->command. execute = ExecuteTurnOnCommand;
```

这个示例展示了如何使用命令模式来封装请求，使得你可以用不同的请求对客户进行参数化，同时支持撤销操作和日志记录操作。命令模式也有助于将操作的发起者和接收者解耦。
### 8. 状态模式（State Pattern）

**问题**：如何让对象在其内部状态改变时改变其行为？

**示例**：假设你有一个电梯，电梯有不同的状态（如开门、关门、上升、下降等），每个状态下电梯的行为不同。

**解决方案**：状态模式允许对象在其内部状态改变时改变其行为，对象看起来像是改变了其类。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    void (*handle)(void);
} State;

void open_state_handle () {
    printf ("Elevator is open\n");
}

void close_state_handle () {
    printf ("Elevator is closed\n");
}

typedef struct {
    State* state;
} Elevator;

void elevator_request (Elevator* elevator) {
    elevator->state->handle ();
}

int main () {
    State open_state = { .handle = open_state_handle };
    State close_state = { .handle = close_state_handle };

    Elevator elevator = { .state = &open_state };
    elevator_request (&elevator);  // 输出: Elevator is open

    elevator. state = &close_state;
    elevator_request (&elevator);  // 输出: Elevator is closed

    return 0;
}
```

### 9. 迭代器模式（Iterator Pattern）

**问题**：如何顺序访问一个聚合对象中的元素，而不需要暴露其内部表示？

**示例**：假设你有一个书架，里面有很多书。你希望遍历书架中的所有书，而不需要知道书架的内部结构。

**解决方案**：迭代器模式提供了一种方法顺序访问一个聚合对象中的元素，而不需要暴露其内部表示。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int index;
    const char** books;
} Iterator;

int iterator_has_next (Iterator* iterator) {
    return iterator->index < 3;
}

const char* iterator_next (Iterator* iterator) {
    if (iterator_has_next (iterator)) {
        return iterator->books[iterator->index++];
    }
    return NULL;
}

typedef struct {
    const char* books[3];
} BookShelf;

Iterator* book_shelf_iterator (BookShelf* book_shelf) {
    Iterator* iterator = (Iterator*) malloc (sizeof (Iterator));
    iterator->index = 0;
    iterator->books = book_shelf->books;
    return iterator;
}

int main () {
    BookShelf book_shelf = { .books = { "Book 1", "Book 2", "Book 3" } };
    Iterator* iterator = book_shelf_iterator (&book_shelf);

    while (iterator_has_next (iterator)) {
        printf ("%s\n", iterator_next (iterator));
    }
    // 输出:
    // Book 1
    // Book 2
    // Book 3

    free (iterator);
    return 0;
}
```
### 10. 外观模式（Facade Pattern）

### 11. 组合模式（Composite Pattern）

**问题**：如何处理对象的层次结构，使得客户端可以统一处理单个对象和对象组合？

**示例**：假设你有一个文件系统，包含文件和文件夹。文件夹可以包含文件和其他文件夹。你希望统一处理文件和文件夹。

**解决方案**：组合模式允许你将对象组合成树形结构，并像处理单个对象一样处理组合对象。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct Component {
    char* name;
    void (*display)(struct Component*);
} Component;

void component_display (Component* component) {
    printf ("Component: %s\n", component->name);
}

typedef struct {
    Component base;
} File;

typedef struct {
    Component base;
    Component** children;
    int child_count;
} Folder;

void folder_display (Component* component) {
    printf ("Folder: %s\n", component->name);
    Folder* folder = (Folder*) component;
    for (int i = 0; i < folder->child_count; i++) {
        folder->children[i]->display (folder->children[i]);
    }
}

int main () {
    File file1 = { .base = { .name = "file1. txt", .display = component_display } };
    File file2 = { .base = { .name = "file2. txt", .display = component_display } };

    Folder folder1 = { .base = { .name = "Folder1", .display = folder_display }, .children = NULL, .child_count = 0 };
    Folder folder2 = { .base = { .name = "Folder2", .display = folder_display }, .children = NULL, .child_count = 0 };

    folder1. children = (Component**) malloc (sizeof (Component*) * 1);
    folder1. children[0] = (Component*)&file1;
    folder1. child_count = 1;

    folder2. children = (Component**) malloc (sizeof (Component*) * 2);
    folder2. children[0] = (Component*)&file2;
    folder2. children[1] = (Component*)&folder1;
    folder2. child_count = 2;

    folder2.base.display (&folder2. base);
    // 输出:
    // Folder: Folder2
    // Component: file2. txt
    // Folder: Folder1
    // Component: file1. txt

    free (folder1. children);
    free (folder2. children);

    return 0;
}
```
### 12. 桥接模式（Bridge Pattern）

**问题**：如何将抽象部分与实现部分分离，使它们可以独立变化？

**示例**：假设你有一个形状类（如圆形、矩形），每个形状可以有不同的颜色（如红色、蓝色）。你希望形状和颜色可以独立变化。

**解决方案**：桥接模式将抽象部分与实现部分分离，使它们可以独立变化。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    void (*fill)(void);
} Color;

void red_fill () {
    printf ("Filling with Red color\n");
}

void blue_fill () {
    printf ("Filling with Blue color\n");
}

typedef struct {
    Color* color;
    void (*draw)(void);
} Shape;

void circle_draw (Shape* shape) {
    printf ("Drawing Circle with ");
    shape->color->fill ();
}

void rectangle_draw (Shape* shape) {
    printf ("Drawing Rectangle with ");
    shape->color->fill ();
}

int main () {
    Color red = { .fill = red_fill };
    Color blue = { .fill = blue_fill };

    Shape circle = { .color = &red, .draw = (void (*)(void)) circle_draw };
    Shape rectangle = { .color = &blue, .draw = (void (*)(void)) rectangle_draw };

    circle.draw (&circle);  // 输出: Drawing Circle with Filling with Red color
    rectangle.draw (&rectangle);  // 输出: Drawing Rectangle with Filling with Blue color

    return 0;
}
```


### 13. 解释器模式（Interpreter Pattern）

**问题**：如何为简单的语言定义一个语法，并使用该语法解释表达式？

**示例**：假设你有一个简单的数学表达式语言，支持加法和减法。你希望解释并计算这些表达式。

**解决方案**：解释器模式定义了一个语言的文法，并使用解释器来解释语言中的表达式。
```
#include <stdio.h>
#include <stdlib.h>

typedef struct Expression {
    int (*interpret)(struct Expression*);
} Expression;

typedef struct {
    Expression* left;
    Expression* right;
} AddExpression;

int add_expression_interpret (Expression* expression) {
    AddExpression* add_expression = (AddExpression*) expression;
    return add_expression->left->interpret (add_expression->left) + add_expression->right->interpret (add_expression->right);
}

typedef struct {
    Expression* left;
    Expression* right;
} SubtractExpression;

int subtract_expression_interpret (Expression* expression) {
    SubtractExpression* subtract_expression = (SubtractExpression*) expression;
    return subtract_expression->left->interpret (subtract_expression->left) - subtract_expression->right->interpret (subtract_expression->right);
}

typedef struct {
    int value;
} NumberExpression;

int number_expression_interpret (Expression* expression) {
    NumberExpression* number_expression = (NumberExpression*) expression;
    return number_expression->value;
}

int main () {
    NumberExpression number1 = { .value = 10 };
    NumberExpression number2 = { .value = 5 };
    NumberExpression number3 = { .value = 3 };

    AddExpression add_expression = { .left = (Expression*)&number1, .right = (Expression*)&number2 };
    SubtractExpression subtract_expression = { .left = (Expression*)&add_expression, .right = (Expression*)&number3 };

    Expression* expression = (Expression*)&subtract_expression;
    int result = expression->interpret (expression);
    printf ("Result: %d\n", result);  // 输出: Result: 12

    return 0;
}
```
