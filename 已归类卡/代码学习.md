void MidDcMotorInMainParaGet(DCMOTOR_INMAIN_INDEX MotorId, MOTOR_PARA ParaType, void *Para)
{
    DC_MANAGER *ptManager = MidGetDcMotorInMainManagerInstance(MotorId);
    U32 DataTemp = 0;
    switch (ParaType)
    {
        case MOTOR_PARA_TARSPEED:   QueryvoidPara(ptManager->mTargetSpeed,U16);         break;
        case MOTOR_PARA_TARSTEP:    QueryvoidPara(ptManager->mTarStep,U32);             break;
        case MOTOR_PARA_STARTPARA:  QueryvoidPara(ptManager->mStartPwmValue,U16);       break;
        case MOTOR_PARA_DIR:        QueryvoidPara(ptManager->mDir,U8);                  break;
        case MOTOR_PARA_OSSEM:      *(OS_SEM **)Para = ptManager->mptSem;               break;
        case MOTOR_PARA_LENPERSTEP: 
        {
            U32 LenPerStep;
            if (ptManager->mCaptureMode == FPGA_CAPTURE_MODE_HIGH || ptManager->mCaptureMode == FPGA_CAPTURE_MODE_LOW)
            {
                LenPerStep = ptManager->mLenPerStep * 2;
            }
            else
            {
                LenPerStep = ptManager->mLenPerStep;
            }
            QueryvoidPara(LenPerStep, U32);
            break;
        }    
        case MOTOR_PARA_CURRSTEP:   QueryvoidPara(ptManager->mCurStep,U32);             break;
        case MOTOR_PARA_CURRSPEED:  
        {
            DataTemp = ptManager->mCurrentSpeed;
            QueryvoidPara(DataTemp,U32);
            break;
        }
        case MOTOR_PARA_CURRPWM:    QueryvoidPara(ptManager->mPwmValue,U16);            break;
        case MOTOR_PARA_STATE:      QueryvoidPara(ptManager->mState,MOTOR_RUNSTATE);    break;
        case MOTOR_PARA_STEPRECORD: QueryvoidPara(DrvGetDcMotorInMainRecordValue(MotorId),U32);  break;
        case MOTOR_PARA_SPEEDCONST: QueryvoidPara(ptManager->SpeedConstStep,U32);       break;
        case MOTOR_PARA_CALLBACK:   QueryvoidPara(ptManager->mIntCallBack,U32);       break;
        case MOTOR_PARA_ERRCODE:    QueryvoidPara(ptManager->mErrCode,U32);       break;
        case MOTOR_PARA_PID:
        {
            PID_PARA_CONFIG PidParaCfg;
            PidParaCfg.mP = (U32)(ptManager->mPidPara.mPPara * 1000);
            PidParaCfg.mI = (U32)(ptManager->mPidPara.mIPara * 1000);
            PidParaCfg.mD = (U32)(ptManager->mPidPara.mDPara * 1000);
            memcpy((void *)(Para), &PidParaCfg, sizeof(PID_PARA_CONFIG));
        }
        break;
        case MOTOR_PARA_HOLDTIME:
        default:
        break;
    }
}


#define QueryvoidPara(d,t)  (*(t *)(Para) = (d))
