<%
// autoMoveByUpProperty. js

function autoMoveByUpProperty (tp) {
    // --- 配置区 ---
    // 请将这里的文件夹路径修改为你自己的目标文件夹
    const targetFolderPath = "已归类文件夹"; 
    // --- 配置区结束 ---

    // 注册一个一次性的加载事件，确保在Obsidian完全加载后再执行我们的脚本
    app.workspace.onLayoutReady (async () => {
        console.log ("Templater - 自动归档脚本：正在注册文件创建监控...");

        // 检查并创建目标文件夹（如果不存在）
        if (! await app.vault.adapter.exists (targetFolderPath)) {
            console.log (`Templater - 自动归档脚本：目标文件夹 "${targetFolderPath}" 不存在，正在创建...`);
            await app.vault.createFolder (targetFolderPath);
        }

        // 注册文件创建事件监听器
        app.vault.on ('create', async (file) => {
            // 确保我们只处理 Markdown 文件
            if (! (file instanceof TFile) || !file.path.endsWith ('. md')) {
                return;
            }

            // 短暂延迟，以确保 frontmatter 已被写入并且元数据缓存已更新
            // 这是必要的，因为 `create` 事件触发时，文件内容可能尚未完全写入
            await sleep (500); // 延迟 500 毫秒

            try {
                // 获取文件的元数据缓存
                const fileCache = app.metadataCache.getFileCache (file);
                
                // 检查 frontmatter 中是否存在 'up' 属性
                // fileCache?. frontmatter?. up 是一种安全的写法，避免因不存在 frontmatter 而报错
                if (fileCache?. frontmatter?. up) {
                    console.log (`检测到笔记 "${file. basename}" 包含 'up' 属性，准备移动...`);
                    
                    const newPath = `${targetFolderPath}/${file. name}`;
                    
                    // 移动文件
                    await app.fileManager.renameFile (file, newPath);
                    console.log (`笔记 "${file. basename}" 已成功移动到 "${newPath}"`);
                    
                    // (可选) 如果你想在移动后弹出一个提示，可以取消下面的注释
                    // new Notice (`笔记 "${file. basename}" 已归档`);
                }
            } catch (e) {
                console.error ("Templater - 自动归档脚本出错: ", e);
                // new Notice ("自动归档脚本出错，请检查开发者控制台。");
            }
        });
    });
}

// 导出函数，以便 Templater 可以调用它
module. exports = autoMoveByUpProperty;
%>
