<%
module.exports = function classifyFile() {

    // 使用 setTimeout 将执行推迟到下一个事件循环

    // 这可以避免在脚本执行期间与其他插件或UI更新冲突

    setTimeout(async () => {

        // --- 配置 ---

        const destinationFolder = "已归类卡";

  

        // 1. 获取当前活动文件

        const file = app.workspace.getActiveFile();

        if (!file) {

            new Notice("错误：无法获取当前文件。");

            console.error("无法获取当前活动文件。");

            return;

        }

  

        console.log(`[File Classifier] 正在处理文件: ${file.name}`);

  

        // 2. 获取文件的元数据

        const fileCache = app.metadataCache.getCache(file.path);

        const frontmatter = fileCache?.frontmatter;

  

        console.log("[File Classifier] Frontmatter:", frontmatter);

  

        // 3. 检查 'up' 属性

        if (frontmatter && 'up' in frontmatter) {

            console.log("[File Classifier] 找到 'up' 属性。");

  

            const currentFolder = file.parent.path;

            const fileName = file.name;

  

            // 4. 避免重复移动

            if (currentFolder === destinationFolder) {

                new Notice(`文件已在 "${destinationFolder}" 中。`);

                return;

            }

  

            // 5. 确保目标文件夹存在

            if (!await app.vault.adapter.exists(destinationFolder)) {

                await app.vault.createFolder(destinationFolder);

                new Notice(`已创建文件夹: ${destinationFolder}`);

            }

  

            // 6. 执行移动

            const newPath = `${destinationFolder}/${fileName}`;

            try {

                await app.fileManager.renameFile(file, newPath);

                new Notice(`文件已归类到 -> ${destinationFolder}`);

                console.log(`[File Classifier] 成功移动文件到: ${newPath}`);

            } catch (e) {

                new Notice("文件移动失败，请检查控制台。");

                console.error("[File Classifier] 文件移动失败:", e);

            }

        } else {

            new Notice("未找到 'up' 属性，不执行移动。");

            console.log("[File Classifier] 未找到 'up' 属性。");

        }

    }, 100); // 100毫秒的延迟

}

%>
