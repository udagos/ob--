<%
// --- 用户配置区 ---

  

// 1. 目标文件夹名称

// 当笔记满足条件时，将会被移动到这个文件夹。

const destinationFolder = "已归类卡";

  

// 2. 触发移动的元数据属性名

// 脚本会检查笔记的元数据中是否存在这个键（key）。

const triggerProperty = "up";

  

// --- 脚本核心代码 ---

  

/**

 * 根据笔记的 frontmatter（元数据）自动移动文件。

 * @returns {Promise<void>}

 */

async function moveNoteBasedOnFrontmatter() {

    try {

        // 步骤 1: 获取当前在编辑器中打开的文件

        const file = app.workspace.getActiveFile();

        // 如果没有打开任何文件，则提示并退出

        if (!file) {

            new Notice("错误：没有活动的笔记文件。");

            return;

        }

  

        // 步骤 2: 获取文件的元数据缓存

        const frontmatter = app.metadataCache.getFileCache(file)?.frontmatter;

  

        // 步骤 3: 检查元数据是否存在，并且是否包含指定的触发属性

        if (frontmatter && triggerProperty in frontmatter) {

            // 步骤 4: 获取当前文件的路径信息

            const currentFolder = file.parent.path;

            const fileName = file.name;

            // 步骤 5: 检查文件是否已经在目标文件夹中，避免重复操作

            if (currentFolder === destinationFolder) {

                new Notice(`文件已在 "${destinationFolder}" 中，无需移动。`);

                return;

            }

            // 步骤 6: 检查目标文件夹是否存在，如果不存在则创建

            const destinationExists = app.vault.getAbstractFileByPath(destinationFolder);

            if (!destinationExists) {

                await app.vault.createFolder(destinationFolder);

                new Notice(`已创建文件夹: ${destinationFolder}`);

            }

            // 步骤 7: 构建新的文件路径并执行移动操作

            const newPath = `${destinationFolder}/${fileName}`;

            await app.fileManager.renameFile(file, newPath);

            new Notice(`文件已成功归类到 -> ${destinationFolder}`);

  

        } else {

            // 步骤 8: 如果未找到触发属性，则明确提示用户

            new Notice(`未找到 "${triggerProperty}" 属性，不执行移动。`);

        }

    } catch (error) {

        // 步骤 9: 捕获并报告任何在执行过程中发生的未知错误

        console.error("笔记归档脚本发生错误:", error);

        new Notice("脚本执行出错，详情请查看开发者控制台。");

    }

}

  

// --- 执行脚本 ---

moveNoteBasedOnFrontmatter();
%>
