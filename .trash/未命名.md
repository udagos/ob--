<%
// --- 配置 ---

// 在这里设置你的目标文件夹名称

const destinationFolder = "已归类卡";

  

// --- 脚本核心 ---

  

async function moveFileBasedOnFrontmatter() {

    // 1. 获取当前活动文件

    const file = app.workspace.getActiveFile();

    if (!file) {

        new Notice("错误：无法获取当前文件。");

        console.error("无法获取当前活动文件。");

        return;

    }

  

    console.log(`正在处理文件: ${file.name}`);

  

    // 2. 获取文件的元数据缓存

    const fileCache = app.metadataCache.getCache(file.path);

    const frontmatter = fileCache?.frontmatter;

  

    // 调试日志：打印 frontmatter 内容

    console.log("Frontmatter:", frontmatter);

  

    // 3. 检查 frontmatter 是否存在，以及是否包含 'up' 属性

    if (frontmatter && 'up' in frontmatter) {

        console.log("找到 'up' 属性。");

  

        // 4. 获取当前文件的路径信息

        const currentFolder = file.parent.path;

        const fileName = file.name;

  

        // 5. 检查文件是否已经在目标文件夹中，避免重复移动

        if (currentFolder === destinationFolder) {

            new Notice(`文件已在 "${destinationFolder}" 中。`);

            console.log(`文件 "${fileName}" 已在目标文件夹中，无需移动。`);

            return;

        }

  

        // 6. 确保目标文件夹存在，如果不存在则创建它

        const destinationExists = await app.vault.adapter.exists(destinationFolder);

        if (!destinationExists) {

            try {

                await app.vault.createFolder(destinationFolder);

                new Notice(`已创建文件夹: ${destinationFolder}`);

                console.log(`创建了目标文件夹: ${destinationFolder}`);

            } catch (e) {

                new Notice("创建目标文件夹失败。");

                console.error("创建目标文件夹失败:", e);

                return;

            }

        }

  

        // 7. 构建新的文件路径并执行移动

        const newPath = `${destinationFolder}/${fileName}`;

        try {

            await app.fileManager.renameFile(file, newPath);

            new Notice(`文件已归类到 -> ${destinationFolder}`);

            console.log(`成功将文件 "${fileName}" 移动到 "${newPath}"`);

        } catch (e) {

            new Notice("文件移动失败，请检查控制台获取详情。");

            console.error("文件移动失败:", e);

        }

  

    } else {

        // 8. 如果没有找到 'up' 属性，则提示用户

        new Notice("未找到 'up' 属性，不执行移动。");

        console.log("未在 frontmatter 中找到 'up' 属性。");

    }

}

  

// 执行函数

moveFileBasedOnFrontmatter();

%>
